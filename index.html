Quick Start
# 1. Environment konfigurieren
cp .env.example .env

# 2. Datenbank + Qdrant + Redis starten
docker compose up -d postgres redis qdrant

# 3. Prisma Migrations + Seed
cd apps/backend && npx prisma migrate dev --name init && npx prisma db seed

# 4. Alle Services starten
docker compose up --build

# 5. Oeffnen:
http://localhost:3000         # Frontend (Next.js)
http://localhost:3001/api/docs # Swagger REST API
http://localhost:8000/docs     # AI Worker FastAPI
http://localhost:6333/dashboard# Qdrant Dashboard
